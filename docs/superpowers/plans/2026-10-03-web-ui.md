# China Salary Calculator Web UI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a cross-platform single-file Web UI (`web/index.html`) that replicates all 7 CLI features with real-time reactive updates using Vue 3.

**Architecture:** Python calculation logic is ported 1:1 to JS; Vue 3 runtime is inlined in a self-contained HTML file; config.json is embedded as default and supports runtime override via file import. Development uses a standalone `compute.js` module + Node.js tests before inlining into the final artifact.

**Tech Stack:** Vue 3 (runtime-only), vanilla CSS, Node.js (dev testing only), SheetJS CDN (optional Excel export), File/Blob APIs.

```javascript
// web/compute.js — China Salary Calculator core engine (ES module for dev, inlined for prod)
// Ported 1:1 from src/salary_calculator.py

export function clamp(v, lo, hi) { return Math.max(lo, Math.min(v, hi)); }
export function roundYuan(v) { return Math.floor(v + 0.5); }
export function fmt(v) { return Number(v).toFixed(2); }

export function taxByTable(taxable, table) {
  if (taxable <= 0) return 0.0;
  for (const [upper, rate, quick] of table) {
    if (taxable <= upper) return taxable * rate - quick;
  }
  return 0.0;
}

export function calcCumulativeTax(cumTaxable, taxBrackets) {
  return taxByTable(cumTaxable, taxBrackets);
}

export function bonusSingleTax(bonus, salary, bonusBrackets, taxFreeMonthly) {
  if (bonus <= 0) return 0.0;
  const base = salary >= taxFreeMonthly ? bonus : Math.max(bonus - (taxFreeMonthly - salary), 0.0);
  if (base <= 0) return 0.0;
  const avg = base / 12.0;
  for (const [upper, rate, quick] of bonusBrackets) {
    if (avg <= upper) return base * rate - quick;
  }
  return base * 0.45 - 181920;
}

export function monthlySpecial(month, items) {
  let total = 0.0;
  for (const { amount, startMonth, endMonth } of items) {
    if (startMonth <= month && month <= endMonth) total += amount;
  }
  return total;
}
```

```javascript
function _normalizeSocial(spec) {
  if (Array.isArray(spec) && spec.length === 2) {
    return { pension: spec, medical: spec, unemployment: spec };
  }
  if (typeof spec === 'object' && !Array.isArray(spec)) {
    const r = {};
    for (const k in spec) r[k] = [...spec[k]];
    return r;
  }
  throw new Error('Invalid social_base spec');
}

export function createSession(cityCode, year, CFG) {
  cityCode = cityCode || CFG.default_city;
  year = parseInt(year || CFG.default_year);
  if (!(cityCode in CFG.cities)) throw new Error(`Unsupported city: ${cityCode}`);
  const city = CFG.cities[cityCode];
  const yearsMap = {};
  for (const k in city.years) yearsMap[parseInt(k)] = city.years[k];
  if (!(year in yearsMap)) throw new Error(`No data for ${cityCode} ${year}`);
  const y = yearsMap[year];
  const sr = city.social_rate;
  const supp = city.supplemental_housing_rate || [0, 0];
  const sess = {
    city_code: cityCode, city_name: city.name, year,
    rent_deduction: parseFloat(city.rent_deduction),
    housing_rate_min: city.housing_rate[0], housing_rate_max: city.housing_rate[1],
    housing_rate_default: city.housing_rate[1],
    extra_min: supp[0], extra_max: supp[1],
    pension_emp: sr.pension_emp, pension_org: sr.pension_org,
    medical_emp: sr.medical_emp, medical_org: sr.medical_org,
    unemploy_emp: sr.unemploy_emp, unemploy_org: sr.unemploy_org,
    injury_org: sr.injury_org || 0.0,
    _year_data: y,
  };

  sess.periodFor = (month) => {
    return month <= y.h1.months[1] ? ['h1', y.h1] : ['h2', y.h2];
  };
  sess.socialBounds = (month) => {
    const [, p] = sess.periodFor(month);
    const s = _normalizeSocial(p.social_base);
    return [s.pension[0], s.pension[1], s.medical[0], s.medical[1], s.unemployment[0], s.unemployment[1]];
  };
  sess.housingBounds = (month) => {
    const [, p] = sess.periodFor(month);
    return [parseFloat(p.housing_base[0]), parseFloat(p.housing_base[1])];
  };
  sess.isProvisional = (month) => {
    const [, p] = sess.periodFor(month);
    return !!p.provisional;
  };
  return sess;
}
```

```javascript
export function computeMonth(sess, monthNo, salary, socialBaseDict, housingBase,
  specialItems, housingPct, extraPct, CFG) {
  const TAX_BRACKETS = CFG.tax_brackets.map(r => [r[0] === null ? Infinity : parseFloat(r[0]), parseFloat(r[1]), parseFloat(r[2])]);
  const TAX_FREE_MONTHLY = parseFloat(CFG.tax_free_monthly);
  const { pension: sbP, medical: sbM, unemployment: sbU } = socialBaseDict;

  const pensionP = sbP * sess.pension_emp;
  const medicalP = sbM * sess.medical_emp;
  const unemployP = sbU * sess.unemploy_emp;
  const housingP = roundYuan(housingBase * housingPct / 100.0);
  const extraP = extraPct ? roundYuan(housingBase * extraPct / 100.0) : 0.0;
  const socialPTotal = pensionP + medicalP + unemployP + housingP + extraP;

  const pensionO = sbP * sess.pension_org;
  const medicalO = sbM * sess.medical_org;
  const unemployO = sbU * sess.unemploy_org;
  const injuryO = sbP * sess.injury_org;
  const housingO = roundYuan(housingBase * housingPct / 100.0);
  const extraO = extraPct ? roundYuan(housingBase * extraPct / 100.0) : 0.0;
  const socialOTotal = pensionO + medicalO + unemployO + injuryO + housingO + extraO;

  const specialM = monthlySpecial(monthNo, specialItems);
  const cumIncome = salary * monthNo;
  const cumBase = TAX_FREE_MONTHLY * monthNo;
  const cumSocial = socialPTotal * monthNo;
  const cumSpecial = specialM * monthNo;
  const cumTaxable = cumIncome - cumBase - cumSocial - cumSpecial;
  const cumTax = calcCumulativeTax(cumTaxable, TAX_BRACKETS);
  const prevCumTaxable = cumTaxable - salary + TAX_FREE_MONTHLY + socialPTotal + specialM;
  const prevCumTax = calcCumulativeTax(prevCumTaxable, TAX_BRACKETS);
  const monthTax = Math.max(0.0, cumTax - prevCumTax);
  const net = salary - socialPTotal - monthTax;

  let rate = 0.0, quick = 0.0;
  for (const [upper, r, q] of TAX_BRACKETS) {
    if (cumTaxable <= upper) { rate = r; quick = q; break; }
  }

  return {
    month_no: monthNo, salary, sb_p: sbP, sb_m: sbM, sb_u: sbU, housing_base: housingBase,
    pension_p: pensionP, medical_p: medicalP, unemploy_p: unemployP,
    housing_p: housingP, extra_p: extraP, social_p_total: socialPTotal,
    pension_o: pensionO, medical_o: medicalO, unemploy_o: unemployO,
    injury_o: injuryO, housing_o: housingO, extra_o: extraO, social_o_total: socialOTotal,
    cum_taxable: cumTaxable, cum_tax: cumTax, month_tax: monthTax,
    rate, quick, net, special_monthly: specialM, housing_pct: housingPct, extra_pct: extraPct,
  };
}

export function computeYear(sess, salary, specialItems, housingPct, extraPct, CFG) {
  const results = [];
  for (let m = 1; m <= 12; m++) {
    const [plo, phi, mlo, mhi, ulo, uhi] = sess.socialBounds(m);
    const [hlo, hhi] = sess.housingBounds(m);
    const sb = { pension: clamp(salary, plo, phi), medical: clamp(salary, mlo, mhi), unemployment: clamp(salary, ulo, uhi) };
    const hb = clamp(salary, hlo, hhi);
    results.push(computeMonth(sess, m, salary, sb, hb, specialItems, housingPct, extraPct, CFG));
  }
  return results;
}
```

```javascript
export function compareBonus(salary, socialPMonthly, specialMonthly, bonus, CFG) {
  const TAX_BRACKETS = CFG.tax_brackets.map(r => [r[0] === null ? Infinity : parseFloat(r[0]), parseFloat(r[1]), parseFloat(r[2])]);
  const BONUS_BRACKETS = CFG.bonus_brackets.map(r => [r[0] === null ? Infinity : parseFloat(r[0]), parseFloat(r[1]), parseFloat(r[2])]);
  const TAX_FREE_MONTHLY = parseFloat(CFG.tax_free_monthly);

  const annualIncome = salary * 12;
  const annualBase = TAX_FREE_MONTHLY * 12;
  const annualSocial = socialPMonthly * 12;
  const annualSpecial = specialMonthly * 12;

  const salaryTaxA = calcCumulativeTax(annualIncome - annualBase - annualSocial - annualSpecial, TAX_BRACKETS);
  const bonusTaxA = bonusSingleTax(bonus, salary, BONUS_BRACKETS, TAX_FREE_MONTHLY);
  const totalA = salaryTaxA + bonusTaxA;

  const taxableB = annualIncome + bonus - annualBase - annualSocial - annualSpecial;
  const totalB = taxByTable(taxableB, TAX_BRACKETS);
  return { totalA, totalB, salaryTaxA, bonusTaxA, taxableB };
}

export function estimateAnnualSettlement(months12, CFG, annualMedical = 0.0) {
  const TAX_BRACKETS = CFG.tax_brackets.map(r => [r[0] === null ? Infinity : parseFloat(r[0]), parseFloat(r[1]), parseFloat(r[2])]);
  const TAX_FREE_MONTHLY = parseFloat(CFG.tax_free_monthly);
  const annualGross = months12.reduce((s, m) => s + m.salary, 0);
  const annualContrib = months12.reduce((s, m) => s + m.social_p_total, 0);
  const annualSpecial = months12.reduce((s, m) => s + m.special_monthly, 0);
  const annualTaxable = Math.max(0.0, annualGross - TAX_FREE_MONTHLY * 12 - annualContrib - annualSpecial - Math.max(0.0, annualMedical));
  const annualSettlementTax = calcCumulativeTax(annualTaxable, TAX_BRACKETS);
  const withheldTax = months12[11].cum_tax;
  const estimatedRefund = withheldTax - annualSettlementTax;
  return { annualTaxable, annualSettlementTax, withheldTax, estimatedRefund };
}

export function inverseGrossFromNet(sess, targetNet, specialItems, housingPct, extraPct, CFG, tol = 0.01, maxIter = 120) {
  let lo = 0.0, hi = 10_000_000.0;
  const targetTotal = targetNet * 12.0;
  let mid, months, netTotal;
  for (let i = 0; i < maxIter; i++) {
    mid = (lo + hi) / 2.0;
    months = computeYear(sess, mid, specialItems, housingPct, extraPct, CFG);
    netTotal = months.reduce((s, m) => s + m.net, 0);
    if (Math.abs(netTotal - targetTotal) <= tol) break;
    if (netTotal < targetTotal) lo = mid; else hi = mid;
  }
  const taxTotal = months[11].cum_tax;
  const contribTotal = months.reduce((s, m) => s + m.social_p_total, 0);
  return { gross: mid, net_12m: netTotal, tax_12m: taxTotal, contrib_12m: contribTotal, months };
}
```

```javascript
function importConfig(file) {
  const reader = new FileReader();
  reader.onload = (e) => {
    try {
      const external = JSON.parse(e.target.result);
      deepMerge(CFG, external);
      app.config.globalProperties.$forceUpdate();
      document.getElementById('import-status').textContent =
        `已加载外部配置：${file.name}（${Object.keys(CFG.cities).length} 城市）`;
    } catch (err) { alert('JSON 解析失败：' + err.message); }
  };
  reader.readAsText(file);
}

function deepMerge(target, source) {
  for (const key of Object.keys(source)) {
    if (source[key] && typeof source[key] === 'object' && !Array.isArray(source[key])) {
      if (!target[key]) target[key] = {};
      deepMerge(target[key], source[key]);
    } else {
      target[key] = source[key];
    }
  }
}
```

```javascript
  ['当月个税','month_tax'],['实发到手','net'],
];

function exportCSV() {
  const rows = months12.value.map(m => {
    const hk = m.month_no <= 6 ? 'H1' : 'H2';
    const obj = { ...m, half: hk };
    return CSV_COLUMNS.map(([_, key]) => obj[key] ?? '');
  });
  const header = CSV_COLUMNS.map(([display]) => display).join(',');
  const body = rows.map(r => r.join(',')).join('\n');
  const blob = new Blob(['\uFEFF' + header + '\n' + body], { type: 'text/csv;charset=utf-8' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url; a.download = `salary_${params.city}_${params.year}.csv`;
  a.click(); URL.revokeObjectURL(url);
}
```

```javascript
function exportExcel() {

  script.src = 'https://cdn.jsdelivr.net/npm/xlsx@0.18.5/dist/xlsx.full.min.js';

function doExcel() {
  const data = months12.value.map(m => {
    const hk = m.month_no <= 6 ? 'H1' : 'H2';

    return row;
  });
  const ws = XLSX.utils.json_to_sheet(data);
  const wb = XLSX.utils.book_new();
  XLSX.utils.book_append_sheet(wb, ws, '逐月明细');
```
